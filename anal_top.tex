\subsection{Real Analysis}

\subsubsection{Continuity (on metric spaces)}

Recall the following notions of continuity.

\begin{df}
    Let $(X, d_X)$ and $(Y, d_Y)$ be metric spaces, and $f \colon X \rightarrow Y$ be a map. Then, 
    \begin{itemize}
        \item We say that $f$ is \emph{continuous} at $x \in X$, 
        if for every $\epsilon > 0$ there is $\delta > 0$ such that for every $x' \in X$ 
        satisfying $d_X(x, x') < \delta$, we have 
        \[
        d_Y(f(x), f(x')) < \epsilon.
        \]
        $f$ is \emph{continuous} if $f$ is continuous at every $x \in X$.
        \item We say that $f \colon X \rightarrow Y$ is \emph{uniformly continuous}, 
        if $f$ is continuous at every $x \in X$ and $\delta = \delta (\epsilon)$ doesn't depend on $x$.
        Let $(X, d_X)$ and $(Y, d_Y)$ be metric spaces, and $f \colon X \rightarrow Y$. 
        \item We say that $f$ is \emph{Lipschitz}, if there is a constant $M > 0$ such that for all $x_1, x_2 \in X$, we have
        \[
        d_Y(f(x_1), f(x_2)) \leq M \cdot d_X(x_1, x_2).
        \]
        \item We say that $f$ is \emph{bi-Lipschitz}, if there are constants $M_1 > 0, M_2 > 0$ such that for all $x_1, x_2 \in X$, we have 
        \[
        M_2 \cdot d_X (x_1, x_2) \leq d_Y(f(x_1), f(x_2)) \leq M_1 \cdot d_X (x_1, x_2).
        \]
        \item We say that $f$ is an \emph{isometry}, or \emph{distance preserving}, if for every $x_1, x_2 \in X$, we have 
        \[
        d_Y(f(x_1), f(x_2)) = d_X(x_1, x_2).
        \]
    \end{itemize}
\end{df}

\begin{remark}
It is immediate that we have the following implications:
\[
\text{biLipschitz} \implies \text{Lipschitz} \implies \text{Uniform Continuity} \implies \text{Continuity}
\]
\end{remark}

\begin{theorem}
A continuous function $f \colon S \rightarrow \R$, where $S \subseteq \R^n$ is compact, 
is uniformly continuous.
\end{theorem}

\begin{proposition}
BiLipschitz maps are injective.
\end{proposition}

\begin{proposition}
A surjective biLipschitz map is a homeomorphism.
\end{proposition}

\begin{theorem}
Let $X$ be a complete metric space. If $A \subseteq X$ is closed, then $A$ is also complete.
\end{theorem}

\subsubsection{Convergence of Functions, DCT}

Recall the following different notions of convergence of functions:

\begin{df}
A sequence of functions $f_n \colon [a, b] \rightarrow \mathbb{R}$ 
\emph{converges point-wise} to $f \colon [a,b] \rightarrow \mathbb{R}$,
if for every $x \in [a,b]$, the sequence of real numbers $f_n(x)$ converges to $f(x)$. 
That is, for every $x \in [a,b]$ and every $\epsilon > 0$, there exists $N_{x, \epsilon} \in \mathbb{N}$ 
such that for all $n \geq N_{x, \epsilon}$ we have $|f_n(x) - f(x)| < \epsilon$.
\end{df}

\begin{df}
A sequence of functions $f_n \colon [a, b] \rightarrow \mathbb{R}$ 
\emph{converges uniformly} to $f \colon [a, b] \rightarrow \mathbb{R}$, 
if for all $\epsilon > 0$, there exists $N_{\epsilon} > 0$ such that for all $n \geq N_{\epsilon}$ and for all $x \in [a, b]$ we have $|f_n(x) - f(x)| < \epsilon$. This is equivalent to
\[
\sup \limits_{x \in [a, b]} |f_n(x) - f(x)| \rightarrow 0, \text{ as } n \rightarrow \infty.
\]
\end{df}

\begin{remark}
Note that it is clear that uniform convergence implies point-wise convergence.
\end{remark}

\begin{remark}
If we have uniform convergence, we can interchange limits and integrals. 
Point-wise convergence is generally not enough. For point-wise convergence, we can 
genrerally use DCT.
\end{remark}

\begin{theorem}[Dominated Convergence Theorem]
Let $g \colon X \rightarrow [0, \infty]$ be integrable,
and $f, f_n \colon X \rightarrow [-\infty, \infty]$, 
$n \geq 1$ be measurable functions such that 
\[
|f_n(x)| \leq g(x), \forall x \in X \text{ and } f_n \overset{n \rightarrow \infty}{\rightarrow} f \mu\text{-a.e.} 
\]
Then,
\[
\left| \int f_n d\mu - \int f d\mu \right| \leq \int |f_n - f| d\mu \overset{n \rightarrow \infty}{\rightarrow} 0.
\]
\end{theorem}

\begin{remark}
$f_n \rightarrow f$ $\mu$-a.e. means that $f_n$ tends to $f$ pointwise 
except for a measure zero set.
\end{remark}


\subsection{Supremum, Infimum, Limsup, and Liminf}

Recall that for a real sequence $(a_n)$, we can define
\[
\limsup_{n\to\infty}a_n := \lim_{n\to\infty}\sup_{k\ge n}a_k
\]
and
\[
\liminf_{n\to\infty}a_n:=\lim_{n\to\infty}\inf_{k\ge n}a_k.
\]
We recall some useful properties regarding $\limsup$ and $\liminf$.

\begin{proposition}[Sequences of Reals]
Let $(a_n)$ be a real sequence. Then, the following hold:
\begin{itemize}
\item $\liminf a_n \leq \limsup a_n$
\item $a_n$ converges to $L$ if and only if $\liminf a_n=\limsup a_n=L$.
\item $\limsup(-a_n)=-\liminf a_n$ and $\liminf(-a_n)=-\limsup a_n$.
\item If $a_n$ is bounded, then \begin{itemize}
    \item $\limsup a_n$ is the largest subsequential limit.
    \item $\liminf a_n$ is the smallest subsequential limit.
\end{itemize}
\end{itemize}
\end{proposition}

\begin{proposition}[Sequences of Sets]
Let $(A_n)$ be a sequence of sets. Then,
\[
\limsup_{n\to\infty}A_n = \bigcap_{N=1}^{\infty} \bigcup_{n\ge N}A_n,
\]
which is the set of points belonging to infinitely many $A_n$. We also have 
\[
\liminf_{n\to\infty}A_n = \bigcup_{N=1}^{\infty}\bigcap_{n\ge N}A_n,
\]
which is the set of points belonding to all but finitely many $A_n$.
\end{proposition}


\subsection{Topology}

\subsubsection{Open and Closed Sets}

\begin{lemma}
    Let $Y$ be a subspace of $X$. If $U$ is open in $Y$ and $Y$ is open in $X$, then
    $U$ is open in $X$.
\end{lemma}

\begin{remark}
    Note that if $Y$ is not open in $X$, then there is no reason $U$ should be open in $X$.
    As a counter example, consider $X = \R$ with the standard topology, $Y = [0,1]$ (with the 
    subpace topology), and $U = [0,\frac{1}{2})$. $U = (-\frac{1}{2}, \frac{1}{2}) \cap Y$, 
    so is open in $Y$. However, clearly, $U$ is not open in $X$, since opens in $X$ are unions
    of open intervals.
\end{remark}

\begin{example}
In the discrete topology on a set $X$, every subset of $X$ is open.
It follows that every set is closedE as well. 
It follows that every function from a discrete space is continuous.
\end{example}

\begin{example}
    In any topology space $X$, $\emptyset$ and $X$ are both closed and open.
\end{example}

A nice way to test if a set is closed in the subspace topology of some 
larger set is from the following:

\begin{theorem}
    Let $Y \subseteq X$. $A$ is closed in $Y$ if and only if it equals 
    the intersection of a closed set of $X$ with $Y$.
\end{theorem}

It also follows then, from the lemma earlier, that:
\begin{corollary}
    Let $Y \subseteq X$. If $A$ is closed in $Y$ and $Y$ is closed in $X$, 
    then $A$ is closed in $X$.
\end{corollary}

There was also the notion of a closure, which is the intersection of all closed
sets containing a set. Note that the resulting set is closed because arbitrary 
intersections of closed sets are closed. 

\begin{theorem}
    Let $A \subseteq X$ be (only) a subset.
    \begin{itemize}
        \item $x \in \overline{A}$ if and only if every open set $U$ containing $x$
            intersects $A$.
        \item If the topology of $X$ is given by a basis, then $x \in \overline{A}$ 
        if and only if every basis element containing $x$ intersects $A$.
    \end{itemize}
\end{theorem}

\begin{exercise}
    Let $X = \{n \in \Z \colon n \geq 2\}$ be the topoological space $X$ with the topology 
    generated by sets of the form $U_k := \{n \in X \colon n \mid k\}$, for $k \geq 2$, so that
    open sets of $X$ are the empty set and arbitrary unions of the sets $U_k$. For any $n \in X$,
    determine the closure of the one point set $\{n\}$ in $X$.
\end{exercise}
\begin{solution}
    Note that $m \in \overline{\{n\}}$ if and only if every open neighborhood of $m$ meets
    $\{n\}$. In other words, we mean that if $m \in U_k$, then $\{n\} \cap U_k \neq \emptyset$,
    so $n \in U_k$. That is to say, if $m \mid k$, then we need $n \mid k$. Therefore, $m = ns$.
    Therefore, the closure is of the form $\{sn \colon s \in \Z_{>0}\}$.
\end{solution}

\begin{df}
    If $A$ is a subset of the topological space $X$, then $x \in X$ is called a \emph{limit point}
    of $A$ if every neighborhood of $x$ intersects $A$ in some point other than $x$ itself.
\end{df}

\begin{remark}
    In other words, $x$ is a limit point of $A$ if $x$ is in the closure of $A \setminus \{x\}$.
\end{remark}

\begin{theorem}
The closure of a set $A$ in a topological space $X$ is precisely the set $A$ union with
the set of limit points of $A$ in $X$.
\end{theorem}

\begin{remark}
A useful thing to note is that $\overline{A} = A \cup \{\text{limit points}\}$ and 
$\partial A = \overline{A} \setminus A^{\circ}$.
\end{remark}

\subsubsection{Hausdorff}

Recall the notion of Hausdorff:

\begin{df}
    A topological space $X$ is \emph{Hausdorff} if for each pair of distinct 
    points $x,y \in X$, there are neighborhoods of $x,y$, respectively, 
    which are disjoint.
\end{df}

\begin{theorem}
    Every finite point set in a Hausdorff space is closed.
\end{theorem}

\begin{theorem}
    If $X$ is a Hausdorff space, then a sequence of points of $X$ converges 
    to at most one point of $X$.
\end{theorem}

\begin{proposition}
    Products and subspaces of Hausdorff spaces are Hausdorff.
\end{proposition}

\begin{proposition}
Metric spaces are Hausdorff.
\end{proposition}

\subsection{Connectedness}

\begin{df}
A topological space $X$ is disconnected if there exists a pair of open sets $U$ and $V$ in $X$ such that
$U \cap V = \emptyset$, $U \cup V = X$, and $U, V \neq \emptyset$. $X$ is connected if it is not disconnected. That is, for all nonempty opens such 
that $U \cap V = \emptyset$ and $U \cup V = X$, $U = \emptyset$ or $V = \emptyset$.
\end{df}

\begin{proposition}
The following are equivalent:
\begin{itemize}
    \item $X$ is connected.
    \item The only subsets of $X$ which are both open and closed are $X$ and $\emptyset$.
    \item $X$ is nonempty and it can't be written as $X = V_1 \cup V_2$, of two disjoint nonempty closed subset.
\end{itemize}
\end{proposition}

\begin{df}
A topological space $X$ is \emph{path connected} if there exists a path between any two points in $X$.
\end{df}

\begin{theorem}
Every path connected space is connected. 
\end{theorem}

\begin{remark}
If the space is locally path connected, then the converse to the above holds.
\end{remark}

\begin{proposition}
The closure of a connected set is connected.
\end{proposition}
\begin{corollary}
Let $X$ be a connected set, and $\overline{X}$ be its closure. 
If $X \subseteq A \subseteq \overline{X}$, then $A$ is also connected.
\end{corollary}

\begin{example}
The Topologist's sine curve is a counter example to the converse statement.
That is, it is connected, but not path connected nor locally path connected.
Here, we are talking about the set $\{(x, \sin (\frac{1}{x})) \colon x \in (0,1]\} \cup \{(0,0)\}$.
If we consider the set 
\[
\{(x, \sin (\frac{1}{x})) \colon x \in (0,1]\} \cup \{(0,y) \colon y \in [-1,1]\},
\]
then we have taken the closure. It remains connected.
\end{example}

\subsection{Compactness}

\begin{df}
A topological space $X$ is \emph{sequentially compact}, if every sequence in $X$ has a 
convergent subsequence in $X$. That is, for every sequence $(x_n)_{n \geq 1}$ in $X$, 
there is a sub-sequence $(x_{n_k})_{k \geq 1}$ and a point $x \in X$ such that $x_{n_k} \rightarrow x$ 
as $k \rightarrow \infty$.
\end{df}

\begin{theorem}
If $X$ is a metric space, then sequentially compactness is equivalent to compactness.
\end{theorem}

\begin{corollary}
Every compact metric space is complete.
\end{corollary}
