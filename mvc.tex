\section{Multivariable Calculus and Geometry}

Here, we include no proofs.

\subsection{2D-Geometry}

\begin{proposition}
Given a circle of radisu $r$ and angle $\theta$ (in radians). Then, 
the area is $\frac{1}{2}r^2 \theta$ and the arc length $r \theta$.
\end{proposition}

\begin{proposition}[Hyperbola]
Consider the hyperbola $\frac{x^2}{a^2}-\frac{y^2}{b^2}=1$. 
Then, the following hold:
\begin{itemize}
\item The foci are $(\pm c, 0)$, where $c^2 = a^2 + b^2$.
\item Its asymptotes are $y = \pm \frac{b}{a}x$, and the angle it constructs with 
the $x$-axis is $\arctan \left(\frac{b}{a}\right)$.
\end{itemize}
\end{proposition}

\begin{proposition}[Ellipses]
Consider the ellipses $\frac{x^2}{a^2}+\frac{y^2}{b^2}=1$. The 
focal distance $c$ satisfies $c^2 = a^2 - b^2$.
\end{proposition}

\begin{proposition}
A tetrahedron with three mutually perpendicular edges of lengths $a,b,c$ 
has volume $\frac{abc}{6}$.
\end{proposition}


\subsection{Trigonometric Identities}

Addition formulae:
\begin{itemize}
\item $\sin(a\pm b)=\sin a\cos b\pm\cos a\sin b$
\item $\cos(a\pm b)=\cos a\cos b\mp\sin a\sin b$
\item $\tan(a\pm b)=\frac{\tan a\pm\tan b}{1\mp\tan a\tan b}$
\end{itemize}

Double and Triple Angles

\begin{itemize}
\item $\sin 2x = 2\sin x\cos x$
\item $\cos 2x=1-2\sin^2x=2\cos^2x-1$
\item $\tan 2x=\frac{2\tan x}{1-\tan^2x}$
\item $\sin 3x=3\sin x-4\sin^3x$
\item $\cos 3x=4\cos^3x-3\cos x$
\item $\tan 3x=\frac{3\tan x-\tan^3x}{1-3\tan^2x}$
\end{itemize}

\subsection{3D Geometry and Basic Limits}

\begin{exercise}
    Find the equation of the plane in $\R^3$ passing through 
    $P=(0,0,1), Q=(1,0,0), R=(1,1,1)$.
\end{exercise}
\begin{solution}
    We use a reference point and use the vector cross product:
    \[
    v := \overrightarrow{PQ} = \langle 1,0,-1\rangle, \qquad 
    w := \overrightarrow{PR} = \langle 1, 1, 0\rangle
    \]
    Now, the normal vector to $P$ is $v \times w$:
    \[
    v \times w = \langle 1,-1,1 \rangle
    \]
    The equation of the plane is given by 
    \[
    \langle 1,-1,1\rangle \cdot \langle x-0, y-0, z-1 \rangle,
    \]
    which is simply $x-y+z = 1$.
\end{solution}

\begin{theorem}[Point-to-Plane Distance]
Let $P=(x_0,y_0,z_0)$ be a point and let $ax+by+cz+d=0$ be 
a plane, where $(a,b,c)\neq(0,0,0)$. Then the distance from $P$ to the plane is
\[
\frac{|ax_0+by_0+cz_0+d|}
{\sqrt{a^2+b^2+c^2}}.
\]
\end{theorem}

\begin{exercise}
    Compute the arc length of the helix $\gamma(t) = (\sin(t),-\cos(t), t)$
    from $t = 0$ to $t = 2\pi$.
\end{exercise}
\begin{solution}
    We must use the arc-length formula $\int_t \lVert \gamma'(t) \rVert dt$.
    \[
    \int_{t = 0}^{2\pi} \sqrt{\cos^2(t) + \sin^2(t) + 1^2} dt 
        = \int_{t = 0}^{2\pi} \sqrt{2} dt = 2\pi \sqrt{2}.
    \]
\end{solution}

The following is a nice way to determine if a limit doesn't exist:
\begin{theorem}
    $\lim \limits_{(x,y) \rightarrow (a,b)} f(x,y) = L$ if and only if for all
    $\gamma \colon [-1,1] \rightarrow \R^2$ such that $\gamma(0) = (a,b)$, then 
    $\lim \limits_{t \to 0} f(\gamma(t)) = L$.
\end{theorem}

\begin{exercise}
Prove that $\lim \limits_{(x,y) \rightarrow (0,0)} \frac{xy}{x^2 + y^2}$
does not exist.
\end{exercise}
\begin{solution}
We consider the paths $\gamma(t) = (t,t)$ and $\delta(t) = (t,-t)$, which both 
tend to $(0,0)$ as $t \rightarrow 0$. Using $\gamma$, we have as follows:
\[
\lim \limits_{(x,y) \rightarrow (0,0)} \frac{xy}{x^2 + y^2} =
\lim \limits_{t \rightarrow 0} \frac{t^2}{t^2 + t^2} = \frac{1}{2}
\]    
On the other hand, $\delta$ gives:
\[
\lim \limits_{(x,y) \rightarrow (0,0)} \frac{xy}{x^2 + y^2} =
\lim \limits_{t \rightarrow 0} \frac{-t^2}{t^2 + t^2} = -\frac{1}{2}
\]    
Therefore, the limit doesn't exist.
\end{solution}

\begin{exercise}
Find $\lim \limits_{(x,y) \rightarrow (0,0)} \frac{xy^2}{x^2 + y^2}$.
\end{exercise}
\begin{solution}
We use polar coordinates and change the limit to $r \to 0$.
\[
\lim \limits_{(x,y) \rightarrow (0,0)} \frac{xy^2}{x^2 + y^2} = 
\lim \limits_{r \rightarrow 0} \frac{r^3\cos \theta \sin^2\theta}{r^2} =
\lim \limits_{r \rightarrow 0} r\cos \theta \sin^2 \theta
\]
Now,
\[
0 \leq |\lim \limits_{r \rightarrow 0} r\cos \theta \sin^2 \theta| \leq 
|\lim \limits_{r \rightarrow 0} r | = 0,
\]
so the limit is $0$ via squeeze.
\end{solution}

\begin{exercise}
Let
\[
z=x^2+y^2,
\qquad
x=t^2,
\qquad
y=e^t.
\]
Find $\dfrac{dz}{dt}$.
\begin{enumerate}[label=(\Alph*)]
    \item $2t^2+2e^t$
    \item $4t+2e^{2t}$
    \item $4t^3+2e^{2t+1}$
    \item $4t^3+2e^{2t}$
    \item $2t^2+e^t$
\end{enumerate}
\end{exercise}
\begin{solution}
(D). This is just chain rule: 
$\frac{dz}{dt} = \Dfrac{z}{x}\Dfrac{x}{t} + \Dfrac{z}{y}\Dfrac{y}{t}$.
\end{solution}

\begin{exercise}
Let
\[
z=f(x,y),
\qquad
x=u^2-v^2,
\qquad
y=2uv.
\]
Which expression gives $\dfrac{\partial z}{\partial u}$?
\begin{enumerate}[label=(\Alph*)]
    \item
    $2u\frac{\partial f}{\partial x}
    +2v\frac{\partial f}{\partial y}$

    \item
    $2u\frac{\partial f}{\partial x}
    -2v\frac{\partial f}{\partial y}$

    \item
    $2v\frac{\partial f}{\partial x}
    +2u\frac{\partial f}{\partial y}$

    \item
    $2u\frac{\partial f}{\partial x}
    +2u\frac{\partial f}{\partial y}$

    \item
    $(u^2-v^2)\frac{\partial f}{\partial x}
    +2uv\frac{\partial f}{\partial y}$
\end{enumerate}
\end{exercise}
\begin{solution}
(A). Again, chain rule.
\end{solution}

\begin{exercise}
Let
\[
z=f(x,y),
\qquad
x=s^2+t,
\qquad
y=st.
\]
Which expression gives $\dfrac{\partial^2z}{\partial s^2}$?
\begin{enumerate}[label=(\Alph*)]
    \item
    $4s^2f_{xx}+4stf_{xy}+t^2f_{yy}$

    \item
    $4s^2f_{xx}+4stf_{xy}+t^2f_{yy}+2f_x$

    \item
    $2f_x+4s^2f_{xx}+4stf_{xy}+2tf_{yy}$

    \item
    $4s^2f_{xx}+2stf_{xy}+t^2f_{yy}$
\end{enumerate}
\end{exercise}
\begin{solution}
(B). First, we find 
\[
z_s = f_x \cdot x_s + f_y \cdot y_s 
= 2sf_x + tf_y.
\]
Next, we have
\[
z_{ss} = 2f_x + 2s\pfrac{f_x}{s} + t\pfrac{f_y}{s}.
\]
We have that 
\[
\pfrac{f_x}{s} = f_{xx}x_s + f_{xy}y_s
= 2sf_{xx} + tf_{xy}
\]
and
\[
\pfrac{f_y}{s} = f_{yx}x_s + f_{yy}y_s 
= 2sf_{xy} + tf_{yy}.
\]
Therefore, we have
\[
z_{ss} = 2f_x + 2s(2sf_{xx} + tf_{xy}) + t(2sf_{xy} + tf_{yy}).
\]
This is just (B).
\end{solution}

\subsection{Curvature}

\begin{df}
Let $I\subseteq\mathbb{R}$ be an open interval, and let $r:I\to\mathbb{R}^n$,
where $n\geq 2$, be a $C^2$ regular curve, $r'(t) \neq 0$ for all $t \in I$.
The \emph{unit tangent vector} is defined by
\[
\mathbf{T}(t)
=
\frac{r'(t)}
{\|r'(t)\|}.
\]
The \emph{curvature} of $r$ at $t$
is defined by
\[
\kappa(t)
=
\left\|\frac{d\mathbf{T}}{ds}\right\|
=
\frac{\|\mathbf{T}'(t)\|}
{\|r'(t)\|},
\]
where $s$ denotes arc length, satisfying
\[
\frac{ds}{dt}=\|r'(t)\|.
\]
\end{df}

\begin{theorem}[Curvature of a Plane Curve]
Let $I\subseteq\mathbb{R}$ be an open interval,
and let $t(t) = (x(t), y(t))$ be a $C^2$ regular curve in $\mathbb{R}^2$.
Then its curvature is
\[
\kappa(t)=
\frac{|x'(t)y''(t)-y'(t)x''(t)|}
{(x'(t)^2+y'(t)^2)^{3/2}}.
\]
\end{theorem}

\begin{theorem}[Curvature of a Space Curve]
Let $I\subseteq\mathbb{R}$ be an open interval,
and let $r \colon I \rightarrow \R^3$ be a $C^2$ regular curve. 
Then, its curvature is
\[
\kappa(t)=
\frac{\|\mathbf{r}'(t)\times\mathbf{r}''(t)\|}
{\|\mathbf{r}'(t)\|^3}.
\]
\end{theorem}

\begin{corollary}[Curvature of the Graph of a Function]
Let $I\subseteq\mathbb{R}$ be an open interval and let $f:I\to\mathbb{R}$ be a $C^2$ function.
Consider the plane curve given by the graph
\[
\mathbf{r}(x)=(x,f(x)).
\]
Then the curvature at $(x,f(x))$ is
\[
\kappa(x)=
\frac{|f''(x)|}
{(1+(f'(x))^2)^{3/2}}.
\]
\end{corollary}

\begin{theorem}[Curvature of a Unit-Speed Curve]
Let $I\subseteq\mathbb{R}$ be an open interval
and let $\mathbf{r}:I\to\mathbb{R}^n$
be a $C^2$ curve parametrized by arc length, so $\|r'(s)\| = 1$ for every $s \in I$. Then,
\[
\kappa(s)=\|\mathbf{r}''(s)\|.
\]
\end{theorem}

\begin{theorem}[Curvature of a Circle]
Let $R>0$ and consider the circle $r(t) = (a + R \cos t, b + R \sin t)$, for $t \in \R$. Then, 
its curvature is constant and equal to $\frac{1}{R}$.
\end{theorem}

\begin{exercise}
Consider the plane curve
\[
\mathbf{r}(t)
=
(3\cos t,3\sin t),
\qquad t\in\mathbb{R}.
\]
What is its curvature at $t=\pi/4$?
\begin{enumerate}[label=(\Alph*)]
\item $0$
\item $\frac19$
\item $\frac13$
\item $3$
\item $9$
\end{enumerate}
\end{exercise}
\begin{solution}
(C). This is a circle of radius $3$.
\end{solution}

\begin{exercise}
Let $C$ be the plane curve parametrized by
\[
\mathbf{r}(t)=(t,t^2),
\qquad t\in\mathbb{R}.
\]
What is its curvature at $t=1$?
\begin{enumerate}[label=(\Alph*)]
\item $\frac15$
\item $\frac{2}{5\sqrt5}$
\item $\frac25$
\item $\frac1{\sqrt5}$
\item $2\sqrt5$
\end{enumerate}
\end{exercise}
\begin{solution}
(B). Let $r(t) = (t, t^2, 0)$. Then, 
\[
r'(t) = (1,2t,0), \quad r''(t) = (0,2,0).
\]
Therefore, the curvature is
\[
\frac{|r'(1) \times r''(1)|}{|r'(1)|^3} =  \frac{2}{5^{\frac{3}{2}}}
\]
\end{solution}

\begin{exercise}
Let $C$ be the space curve parametrized by
\[
\mathbf{r}(t)
=
(3\cos t,3\sin t,4t),
\qquad t\in\mathbb{R}.
\]
What is its curvature at $t=0$?
\begin{enumerate}[label=(\Alph*)]
\item $0$
\item $\frac3{25}$
\item $\frac4{25}$
\item $\frac15$
\item $\frac13$
\end{enumerate}
\end{exercise}
\begin{solution}
(B). We have 
\[
r'(t) = (-3 \sin t, 3 \cos t, 4), \qquad r''(t) = (-3\cos t, -3\sin t, 0)
\]
Thus, 
\[
r'(t) \times r''(t) = (-12 \sin t, -12 \cos t, 9).
\]
At $t = 0$, we get $(0,-12,9)$, which has magnitude $15$. At the denominator,
we have $r'(0) = (0,3,4)$, which has magnitude $5$.
\end{solution}


\subsection{Hessian}

Let $f(x,y)$ be a twice-differntiable function on $\R^2$. Then, recall that 
the critical values occur when $f_x$ and $f_y$ are zero. To find the type, 
we use the Hessian matrix, $H$. 
\begin{itemize}
\item If $\det H < 0$, then it is a saddle.
\item If $\det H > 0$ and $f_{xx} > 0$, then it is a relative minima.
\item If $\det H > 0$ and $f_{xx} < 0$, then it is a relative maxima.
\end{itemize}

\subsection{Fourier Transformation}

Recall that how to find the Fourier series of a periodic function.
Let $f(x)$ be defined on the interval $(-L,L)$ and extended outside this
interval by $f(x+2L)=f(x)$, so that $f$ is $2L$-periodic. 
The Fourier series of $f(x)$ is
\[
\frac{a_0}{2}
+
\sum_{n=1}^{\infty}
\left(
a_n\cos\left(\frac{n\pi x}{L}\right)
+
b_n\sin\left(\frac{n\pi x}{L}\right)
\right),
\]
where the Fourier coefficients are
\[
a_n
=
\frac{1}{L}
\int_{-L}^{L}
f(x)\cos\left(\frac{n\pi x}{L}\right)\,dx,
\qquad n=0,1,2,\ldots,
\]
and
\[
b_n
=
\frac{1}{L}
\int_{-L}^{L}
f(x)\sin\left(\frac{n\pi x}{L}\right)\,dx,
\qquad n=1,2,\ldots.
\]
\begin{remark}
Note that in the Fourier series corresponding to an odd function, 
only sine terms can be present. In the Fourier series
corresponding to an even function, only cosine terms (and possibly a constant)
can be present.
\end{remark}
Note that we can also formulate a complex version:
\[
f(x)
\sim
\sum_{n=-\infty}^{\infty}
c_n e^{in\pi x/L},
\]
where the complex Fourier coefficients are
\[
c_n
=
\frac{1}{2L}
\int_{-L}^{L}
f(x)e^{\frac{-in\pi x}{L}}dx.
\]
\begin{exercise}
What is the Fourier series corresponds to the sawtooth wave $s(x)$:
\[
s(x)=
\begin{cases}
x, & x\in[-\pi,\pi),\\
s(x\pm2\pi), & \forall x\in\mathbb{R}.
\end{cases}
\]
\end{exercise}
\begin{solution}
Note that $s(-x) = -x = -s(x)$, so is odd. Thus, $a_0 = 0$ and $a_n = 0$,
so $s(x) = \sum_{n=1}^{\infty} b_n\sin(nx)$. Recall that:
\[
b_n = \frac{1}{\pi} \int_{-\pi}^{\pi} x\sin(nx)dx.
\]
The integrand is even, so 
\[
b_n = \frac{2}{\pi}\int_{0}^{\pi} x\sin(nx)dx,
\]
and integration by parts gives
\[
b_n = \frac{2(-1)^{n + 1}}{n}.
\]
\end{solution}

\subsection{Grad, Div, Curl}

Recall that the gradient satisfies the following:

\begin{proposition}
\[
\nabla \phi = \left\langle\frac{\partial \phi}{\partial x},
\frac{\partial \phi}{\partial y},  \frac{\partial \phi}{\partial z} \right\rangle
\]
Further, if we want the directional derivative of $\phi$ in the deriction of a
unit vector $\hat{s}$, denoted $\pfrac{\phi}{s}$, we have
\[
\pfrac{\phi}{s} = \hat{s} \cdot \nabla \phi.
\]
\end{proposition}


\begin{exercise}
    If $\phi = axy^2 + byz + cx^3z^2$, where $a,b,c$ are constants, find $\nabla \phi$. 
    Further, find the directional derivative of $\phi$ at the point $(1,4,2)$ in the 
    direction towards the point $(2,0-1)$.
\end{exercise}
\begin{solution}
    First, we compute the gradient:
    \[
    \nabla \phi = \left\langle ay^2 + 3cz^2z^2, 2axy + bz, by + 2cx^3z \right\rangle
    \]
    Therefore, the gradient at $P = (1,4,2)$ is:
    \[
    (\nabla \phi)_P = \left\langle 16a + 12c, 8a + 2b, 4b + 4c \right\rangle
    \]
    Now, $s$ is the difference between the two points, which is just
     $s = (2,0,-1) - (1,4,2) = (1,-4,-3)$. Thus, $\hat{s}$ is
    \[
    \hat{s} = \frac{\langle 1, -4 , -3\rangle}{\sqrt{26}}
    \]
    Finally, the directional derivative $\frac{\partial \phi}{\partial s}$, is as follows:
    \[
    \frac{\partial \phi}{\partial s} = (\nabla \phi \cdot \hat{s})_P = \frac{(16a + 12c) - 4(8a + 2b) - 3 (4b + 4c)}{\sqrt{26}} = \frac{-16a - 20b}{\sqrt{26}}
    \]
\end{solution}

\begin{df}
    Let $F$ be a vector field.
    \begin{itemize}
        \item The \emph{divergence} of $F$ is the scalar product 
        $\nabla \cdot F$, writing 
        $\divergence F = \nabla \cdot F$.
        \item The \emph{curl} of $F$ is the vector product 
        $\nabla \times F$, writing $\curl F = \nabla \times F$.
    \end{itemize}
\end{df}

We recall some useful identities relating these different notions.

\begin{theorem}
Let $f,g \colon \R^3 \rightarrow \R$ be a smooth scalar function 
and $A,B$ be a vector field (or more simply, a vector valued function).
    \begin{itemize}
        \item $\curl (\nabla f) = 0$
        \item $\divergence (\curl A) = 0$
        \item $\curl (\curl (A)) = \nabla (\divergence A) - \nabla ^2 A$
        \item $\nabla(fg) = f \nabla g + g \nabla f$
        \item $\divergence (fA) = \nabla f \cdot A + f \divergence \cdot A$
        \item $\curl (fA) = \nabla f \times A + f \curl A$
        \item $\divergence ( A \times B) = B \cdot \curl A - A \cdot \curl B$
    \end{itemize}
\end{theorem}

\begin{exercise}
Let $f,g \colon \R^3 \rightarrow \R$ be smooth scalar functions. 
Show that $\nabla \cdot (\nabla f \times \nabla g) = 0$.
\end{exercise}
\begin{solution}
$\divergence(\nabla f \times \nabla g) = \nabla g (\curl(\nabla f)) - \nabla f(\curl(\nabla g))$.
However, curl of grad is $0$, so each of the parts inside the brackets are zero.
\end{solution}

\begin{exercise}
Let $f,g \colon \R^3 \rightarrow \R$ be smooth scalar functions.   
Show that $\curl (f \nabla g) = \nabla f \times \nabla g$.
\end{exercise}
\begin{solution}
We just apply one of the identities above:
\[
\curl (f \nabla g) = \nabla f \times \nabla g + f \curl (\nabla g),
\]
but curl of grad is zero, so the term on the right vanishes.
\end{solution}

\subsection{Partial Derivatives and Jacobians}

\begin{exercise}
Let $z = e^{xy^2}$, $x = t \cos t$, $y = t \sin t$. Compute $\Dfrac{z}{t}$.
\end{exercise}
\begin{solution}
We use that $\Dfrac{z}{t} = \Dfrac{z}{x} \Dfrac{x}{t} + \Dfrac{z}{y} \Dfrac{y}{t}$.
\begin{align*}
\Dfrac{z}{t} &= y^2e^{xy^2}\cdot (\cos t -t\sin t) + 2xye^{xy^2} \cdot (\sin t + t \cos t) \\
&=t^2 \sin^2 t e^{t^3 \cos t \sin^2t}(\cos t -t\sin t) + 2xye^{t^3 \cos t \sin^2t} \cdot (\sin t + t \cos t)
\end{align*}
\end{solution}

\begin{exercise}
If $x=2r-s$ and $y=r+2s$, find $\frac{\partial^2 U}{\partial y \partial x}$
in terms of derivatives with respect to $r$ and $s$.
\end{exercise}
\begin{solution}
Use the multivariable chain rule, we should really solve for $r$ and $s$ in terms of $x$ and $y$.
We have $5r = 2x + y$ and $5s = 2y - x$. Therefore
\[
\pfrac{U}{x} = \pfrac{U}{r} \pfrac{r}{x} + \pfrac{U}{s} \pfrac{s}{x}
= \frac{2}{5} \pfrac{U}{r} - \frac{1}{5} \pfrac{U}{s}.
\]
We now differentiate by $y$.
\begin{align*}
\frac{\partial^2 U}{\partial y \partial x} &= \pfrac{}{y} \left(\frac{2}{5} \pfrac{U}{r} - \frac{1}{5} \pfrac{U}{s} \right) \\
&= \pfrac{}{r}\left(\frac{2}{5}\frac{\partial^2 U}{\partial r^2} - \frac{1}{5}\frac{\partial^2 U}{\partial rs}\right) \cdot \pfrac{r}{y}
+ \pfrac{}{s}\left(\frac{2}{5}\frac{\partial^2 U}{\partial sr} - \frac{1}{5}\frac{\partial^2 U}{\partial s^2}\right) \cdot \pfrac{s}{y} \\
&= \left(\frac{2}{5}\frac{\partial^2 U}{\partial r^2} - \frac{1}{5}\frac{\partial^2 U}{\partial rs}\right) \cdot \frac{1}{5}
+ \left(\frac{2}{5}\frac{\partial^2 U}{\partial sr} - \frac{1}{5}\frac{\partial^2 U}{\partial s^2}\right) \cdot \frac{2}{5} \\
&= \frac{1}{25} \left(2 \frac{\partial^2 U}{\partial r^2} + 3\frac{\partial^2 U}{\partial rs}
- 2\frac{\partial^2 U}{\partial s^2}\right)
\end{align*}
\end{solution}

\subsection{Line and Surface Integrals}

\begin{df}
Given a vector field $F$, if $\oint_{C} F \cdot d \textbf{r} = 0$ for any 
closed curve $C$, then we say that $\textbf{F}$ is a \textbf{conservative field}.
\end{df}

We recall some nice theorems that help computation.

\begin{theorem}[Green's Theorem in the Plane]
Suppose $R$ is a closed plane region bounded by a simple plane closed convex curve in the $x$-$y$ plane.
Let $L, M$ be continuous functions of $x, y$ having continuous derivatives throughout $R$. Then:
\[
\oint_C (L dx + M dy) = \int_R \left( \frac{\partial M}{\partial x} - \frac{\partial L}{\partial y} \right) dx dy,  
\]
where $C$ is the boundary of $R$ described in the counter-clockwise (positive) sense.
\end{theorem}

\begin{theorem}
Let $\mathbf{A}$ be a vector field (that is continously differentiable on an open, simply 
connected domain $D\subseteq \R^3$). Then, the following conditions are equivalent:
\begin{itemize}
\item $\int_C \mathbf{A} \cdot d\mathbf{r}$ is independent
of the path.
\item There exists a scalar function $\theta$ such that $\mathbf{A} = \nabla \theta$.
In this case, the integral is equal to $\theta(P) - \theta(Q)$, where $P$ and $Q$ denote the finishing and starting points.
\item $\nabla \times \mathbf{A} = 0$ ($\curl \mathbf{A}= 0$).
\item $\mathbf{A}$ is conservative.
\end{itemize}
\end{theorem}

\begin{exercise}
Which of the following apply to the vector field
\[
\vec{F}=\langle x^2-y^2,-2xy,z^2\rangle
\]
defined on all of $\mathbb{R}^3$?

\begin{enumerate}[label=(\Alph*)]
    \item $\operatorname{div}\vec{F}=0$
    \item Conservative and curl vanishes.
    \item Conservative, but curl does not vanish.
    \item Curl vanishes, but not conservative.
\end{enumerate}
\end{exercise}
\begin{solution}
(B). Note that the curl vanishes, and $\R^3$ is simply connected, so (B).
\end{solution}

\begin{df}[Differential Element of Surface Area]
The differential element of surface area is
\[
dS =
\left|
\frac{\partial \mathbf{r}}{\partial v_1}
\times
\frac{\partial \mathbf{r}}{\partial v_2}
\right|
\,dv_1\,dv_2.
\]
\end{df}

\begin{df}
If $S$ is a surface then the \emph{flux} of a vector field $\textbf{A}$ across 
$S$ is defined as
\[
\int_S \textbf{A} \cdot \hat{\textbf{n}} dS,
\]    
where $\hat{\textbf{n}}$ is a unit normal pointing out $S$.
\end{df}

\begin{theorem}[Divergence Theorem]
If $\tau$ is the volume enclosed by a closed surface $S$ with unit outward normal
$\mathbf{n}$ and $\mathbf{A}$ is a continuously differentiable vector field throughout $\tau$,
then 
\[
\int_S \mathbf{A} \cdot \mathbf{n} dS = \int_{\tau} \divergence \mathbf{A} d\tau.
\]
\end{theorem}

\begin{theorem}[Stokes' Theorem]
Suppose $S$ is an open surface with a simple closed curve $C$ forming its boundary, and let $\textbf{A}$ be a vector 
field with continuous partial derivatives. Then:
\[
\oint_{C} \mathbf{A} \cdot d \mathbf{r} 
= \int_S \curl \mathbf{A} \cdot \hat{\mathbf{n}} dS
\]
where the direction of the unit normal to $S$ and the sense of $C$ are related by a right-hand rule. 
That is, $\hat{\mathbf{n}}$ is in the direction a right-handed screw moves when turned in the direction of $C$.
\end{theorem}

\begin{theorem}[Gauss's Flux Theorem]
Let $S$ be a closed surface with outward unit normal $\hat{\textbf{n}}$, and let $O$ be the origin of the coordinate system. Then:
\[
\int_S \frac{\hat{\textbf{n}} \cdot \textbf{r}}{r^3} = \begin{cases}
0, & \text{ if } O \text{ is exterior to } S \\
4\pi, & \text{ if } O \text{ is interior to } S
\end{cases}
\]
\end{theorem}

\begin{exercise}
Evaluate
\[
\int_{(0,1)}^{(1,2)} (x^2-y) dx+(y^2+x) dy
\]
along:
\begin{enumerate}
\item a straight line from $(0,1)$ to $(1,2)$.
\item straight lines from $(0,1)$ to $(1,1)$ and then from $(1,1)$ to $(1,2)$.
\item the parabola $x=t$, $y=t^2+1$.
\end{enumerate}
\end{exercise}
\begin{solution}
1. The line adjoining these is $y = x + 1$. Then, $dy = dx$, and we just plug in:
\[
\int_0^1 (x^2 - (x + 1))dx + ((x+1)^2 + x)dx 
= \int_0^1 2x^2 + 2x 
= \frac{5}{3}.
\]

2. We do a similar thing, but split the integral piecewise. The first part is
\[
\int_0^1 (x^2 - 1)dx + (1 + x)\cdot 0
= -\frac{2}{3}.
\]
The next part is
\[
\int_1^2 (1 - y) \cdot 0 + (y^2 + 1)dy 
= \frac{10}{3}.
\]
Thus, the integral is equal to $\frac{8}{3}$ by linearity. \\

3. For this, we note that $dx = dt$ and $dy = 2t dt$. The time is from $0$ to $1$.
\[
\int_0^1 (t^2 - (t^2 + 1))dt + ((t^2+ 1)^2 + t) 2tdt
= \int_0^1 2t^5 + 4t^2 + 2t^2 + 2t - 1 dt
= 2.
\]
\end{solution}

\begin{exercise}
Evaluate $\int_C yds$ along the curve $C$ given by $y = 2\sqrt{x}$
from $x = 3$ to $x = 24$.
\end{exercise}
\begin{solution}
Recall that $ds$ is the arclength element. It is probably easiest to use that
\[
ds = \sqrt{(dx)^2 + (dy)^2} = \sqrt{1 + \left(\frac{dy}{dx}\right)^2}dx.
\]
We have $\frac{dy}{dx} = \frac{1}{\sqrt{x}}$, so $ds = \sqrt{1 + \frac{1}{x}}dx$.
Thus, the integral is as follows:
\[
\int_3^{24} 2\sqrt{x} \cdot \sqrt{1 + \frac{1}{x}} dx
= 2\int_3^{24} \sqrt{x + 1} dx
= \frac{4}{3}(x + 1)^{\frac{3}{2}}\bigg|_3^{24}
= 156
\]
\end{solution}

\begin{exercise}
Verify Green's theorem in the plane for
\[
\oint_C (2xy-x^2) dx+(x+y^2) dy,
\]
where $C$ is the closed curve of the region bounded by $y=x^2$ and $y^2=x$.
\end{exercise}
\begin{solution}
Note that this is just from the origin to $(1,1)=:P$ and back. From the origin to $P$, we have:
\[
\int_0^1 (2x^3 - x^2)dx + (x + x^4)2xdx
= \int_0^1 2x^5 + 2x^3 + x^2 dx
= \frac{7}{6}.
\]
From $P$ to the origin, we have as follows:
\[
\int_1^0 (2y^3 - y^4) 2ydy + (y^2 + y^2)dy
= \int_1^0 -2y^5 + 4y^4 + 2y^2 dy
= -\frac{17}{15}
\]
Thus, the original integeral is $\frac{1}{30}$. \\

On the other hand, the statement of Green's theorem gives us that the 
integral is equal to the following:
\[
\int_R (1-2x)dxdy.
\]
Now we just need to find the appropriate bounds. We have as follows:
\[
\int_{y = 0}^1 \int_{x = y^2}^{\sqrt{y}} (1-2x)dxdy 
= \int_0^1 \sqrt{y} - y - y^2 + y^4 dy
= \frac{1}{30}.
\]
Thus, true!
\end{solution}

\begin{exercise}
Evaluate
\[
\int_{(1,1)}^{(4,2)} (x+y)\,dx+(y-x)\,dy
\]
along:
\begin{enumerate}
\item the parabola $y^2=x$
\item a straight line
\item straight lines from $(1,1)$ to $(1,2)$ and then to $(4,2)$
\item the curve $x=2t^2+t+1$, $y=t^2+1$
\end{enumerate}
\end{exercise}
\begin{solution}
1. We just plug in simply, using $dx = 2ydy$.
\[
\int_{y = 1}^2 (y^2 + y)2ydy + (y - y^2)dy
= \int_1^2 2y^3 + y^2 + y dy 
= \frac{34}{3}
\]
2. The line connecting the two is $y = \frac{1}{3}(x - 1) + 1 = \frac{1}{3}x + \frac{2}{3}$.
Therefore, $dy = \frac{1}{3} dx$, and the integral is as follows:
\[
\int_{x = 1}^4 (x + \frac{1}{3}x + \frac{2}{3}) dx + (\frac{1}{3}x + \frac{2}{3} - x) \frac{1}{3}dx
= \int_{1}^4 \frac{10}{9}x + \frac{8}{9} dx
= 11
\]
3. We split up the integral. From $(1,1)$ to $(1,2)$, $dx = 0$ and from $(1,2)$
to $(4,2)$, $dy = 0$.
\begin{align*}
\int_{y = 1}^2 (y - 1)dy &= \frac{1}{2} \\
\int_{x = 1}^4 (x + 2)dx &= \frac{27}{2}
\end{align*}
Thus, the total is $14$. \\

4. We parametrize in terms of $t$. Note that the bound is $t = 0$ to $t = 1$.
Also, $dx = (4t + 1)dt$, $dy = 2t dt$.
\[
\int_{t = 0}^1 (3t^2 + t + 2)(4t + 1)dt + (-t^2 - t)(2t)dt
= \int_{t = 0}^1 10t^3 + 5t^2 + 9t + 2
= \frac{32}{3}.
\]
\end{solution}

\begin{exercise}
Let $\mathbf{F}:\mathbb{R}^3\to\mathbb{R}^3$

\begin{exercise}
Let $\mathbf{F}:\mathbb{R}^3\to\mathbb{R}^3$
be given by
\[
\mathbf{F}(x,y,z)=(0,0,x^2).
\]
Let $S$ be the portion of the plane
\[
x+y+z=1
\]
lying in the first octant, oriented with an upward-pointing normal vector.
Let $C=\partial S$ have the induced orientation. Evaluate
\[

\]
\begin{enumerate}[label=(\Alph*)]
\item $-\frac23$
\item $-\frac13$
\item $0$
\item $\frac13$
\item $\frac23$
\end{enumerate}

\begin{solution}
(B). Use Stoke's theorem. Note the outwards unit normal is $\mathbf{n} = \frac{1}{\sqrt{3}}(1,1,1)$. 
We want to make sense of the area element $dS$ first, but for this, we note that our surface can be 
parametrized as $r(x,y) = (x,y, 1 - x - y)$. Thus,

dS = |r_x \times r_y| dx dy = |(1,0,-1) \times (0,1,-1)|dxdy = \sqrt{3}dxdy

\int_S \curl\mathbf{F} \cdot \mathbf{n} dS

\subsection{Jacobians and Volume/ Area Elements}

We include a quick cheat sheet for the volume/ area elements for calculating integrals 
with Euclidean, polar, or spherical coordinates.

\subsubsection{Euclidean (Cartesian) Coordinates}

\begin{example}[$\R^2$]
Here, $x=x$ and $y = y$, so $dA = dxdy$.
\end{example}
\begin{example}[$\R^3$]
Here, $x=x$, $y = y$, and $z = z$, so $dV = dxdydz$.
\end{example}



\subsubsection{Polar Coordinates}

\begin{example}[$\R^2$]
Here, $x = r \cos \theta$ and $y = r \sin \theta$, subject to
\[
x^2 + y^ 2= r^2, \qquad \theta = \arctan \left(\frac{y}{x}\right).
\]
In this case, $dA = r dr d\theta$.
\end{example}

\begin{example}[$\R^3$]
Here, $x = r \cos \theta, y = r \sin \theta, z = z$, subject to
\[
x^2 + y^ 2= r^2, \qquad \theta = \arctan \left(\frac{y}{x}\right).
\]
Here, $dV = rdr d\theta dz$.
\end{example}

\subsubsection{Spherical Coordinates ($\mathbb R^3$)}

Here $\phi$ is measured from the positive $z$-axis and $\theta$ is measured from 
positive $x$-axis to $y$-axis. We set
\[
\begin{aligned}
x &= r\sin\phi\cos\theta,\\
y &= r\sin\phi\sin\theta,\\
z &= r\cos\phi,\\
x^2+y^2+z^2 &= r^2.
\end{aligned}
\]
In this case,
\[
dV=r^2\sin\phi
dr d\phi d\theta
\]
