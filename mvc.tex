\section{Multivariable Calculus}

Here, we include no proofs.

\begin{exercise}
    Find the equation of the plane in $\R^3$ passing through 
    $P=(0,0,1), Q=(1,0,0), R=(1,1,1)$.
\end{exercise}
\begin{solution}
    We use a reference point and use the vector cross product:
    \[
    v := \overrightarrow{PQ} = \langle 1,0,-1\rangle, \qquad 
    w := \overrightarrow{PR} = \langle 1, 1, 0\rangle
    \]
    Now, the normal vector to $P$ is $v \times w$:
    \[
    v \times w = \langle 1,-1,1 \rangle
    \]
    The equation of the plane is given by 
    \[
    \langle 1,-1,1\rangle \cdot \langle x-0, y-0, z-1 \rangle,
    \]
    which is simply $x-y+z = 1$.
\end{solution}

\begin{exercise}
    Compute the arc length of the helix $\gamma(t) = (\sin(t),-\cos(t), t)$
    from $t = 0$ to $t = 2\pi$.
\end{exercise}
\begin{solution}
    We must use the arc-length formula $\int_t \lVert \gamma'(t) \rVert dt$.
    \[
    \int_{t = 0}^{2\pi} \sqrt{\cos^2(t) + \sin^2(t) + 1^2} dt 
        = \int_{t = 0}^{2\pi} \sqrt{2} dt = 2\pi \sqrt{2}.
    \]
\end{solution}

The following is a nice way to determine if a limit doesn't exist:
\begin{theorem}
    $\lim \limits_{(x,y) \rightarrow (a,b)} f(x,y) = L$ if and only if for all
    $\gamma \colon [-,1,1] \rightarrow \R^2$ such that $\gamma(0) = (a,b)$, then 
    $\lim \limits_{t \to 0} f(\gamma(t)) = L$.
\end{theorem}

\begin{exercise}
Prove that $\lim \limits_{(x,y) \rightarrow (0,0)} \frac{xy}{x^2 + y^2}$
does not exist.
\end{exercise}
\begin{solution}
We consider the paths $\gamma(t) = (t,t)$ and $\delta(t) = (t,-t)$, which both 
tend to $(0,0)$ as $t \rightarrow 0$. Using $\gamma$, we have as follows:
\[
\lim \limits_{(x,y) \rightarrow (0,0)} \frac{xy}{x^2 + y^2} =
\lim \limits_{t \rightarrow 0} \frac{t^2}{t^2 + t^2} = \frac{1}{2}
\]    
On the other hand, $\delta$ gives:
\[
\lim \limits_{(x,y) \rightarrow (0,0)} \frac{xy}{x^2 + y^2} =
\lim \limits_{t \rightarrow 0} \frac{-t^2}{t^2 + t^2} = -\frac{1}{2}
\]    
Therefore, the limit doesn't exist.
\end{solution}

\begin{exercise}
Find $\lim \limits_{(x,y) \rightarrow (0,0)} \frac{xy^2}{x^2 + y^2}$.
\end{exercise}
\begin{solution}
We use polar coordinates and change the limit to $r \to 0$.
\[
\lim \limits_{(x,y) \rightarrow (0,0)} \frac{xy^2}{x^2 + y^2} = 
\lim \limits_{r \rightarrow 0} \frac{r^3\cos \theta \sin^2\theta}{r^2} =
\lim \limits_{r \rightarrow 0} r\cos \theta \sin^2 \theta
\]
Now,
\[
0 \leq |\lim \limits_{r \rightarrow 0} r\cos \theta \sin^2 \theta| \leq 
|\lim \limits_{r \rightarrow 0} r | = 0,
\]
so the limit is $0$ via squeeze.
\end{solution}

\subsection{Grad, Div, Curl}

Recall that the gradient satisfies the following:

\begin{proposition}
\[
\nabla \phi = \left\langle\frac{\partial \phi}{\partial x},
\frac{\partial \phi}{\partial y},  \frac{\partial \phi}{\partial z} \right\rangle
\]
Further, if we want the directional derivative of $\phi$ in the deriction of a
unit vector $\hat{s}$, denoted $\pfrac{\phi}{s}$, we have
\[
\pfrac{\phi}{s} = \hat{s} \cdot \nabla \phi.
\]
\end{proposition}


\begin{exercise}
    If $\phi = axy^2 + byz + cx^3z^2$, where $a,b,c$ are constants, find $\nabla \phi$. 
    Further, find the directional derivative of $\phi$ at the point $(1,4,2)$ in the 
    direction towards the point $(2,0-1)$.
\end{exercise}
\begin{solution}
    First, we compute the gradient:
    \[
    \nabla \phi = \left\langle ay^2 + 3cz^2z^2, 2axy + bz, by + 2cx^3z \right\rangle
    \]
    Therefore, the gradient at $P = (1,4,2)$ is:
    \[
    (\nabla \phi)_P = \left\langle 16a + 12c, 8a + 2b, 4b + 4c \right\rangle
    \]
    Now, $s$ is the difference between the two points, which is just
     $s = (2,0,-1) - (1,4,2) = (1,-4,-3)$. Thus, $\hat{s}$ is
    \[
    \hat{s} = \frac{\langle 1, -4 , -3\rangle}{\sqrt{26}}
    \]
    Finally, the directional derivative $\frac{\partial \phi}{\partial s}$, is as follows:
    \[
    \frac{\partial \phi}{\partial s} = (\nabla \phi \cdot \hat{s})_P = \frac{(16a + 12c) - 4(8a + 2b) - 3 (4b + 4c)}{\sqrt{26}} = \frac{-16a - 20b}{\sqrt{26}}
    \]
\end{solution}

\begin{df}
    Let $F$ be a vector field.
    \begin{itemize}
        \item The \emph{divergence} of $F$ is the scalar product 
        $\nabla \cdot F$, writing 
        $\divergence F = \nabla \cdot F$.
        \item The \emph{curl} of $F$ is the vector product 
        $\nabla \times F$, writing $\curl F = \nabla \times F$.
    \end{itemize}
\end{df}

We recall some useful identities relating these different notions.

\begin{theorem}
    Let $\phi \colon \R^3 \rightarrow \R$ and $F$ be a vector field (or 
    more simply, a vector valued function).
    \begin{itemize}
        \item $\curl (\nabla \phi) = 0$
        \item $\divergence (\curl F) = 0$
        \item $\curl (\curl (F)) = \nabla (\divergence F) - \nabla ^2 F$
    \end{itemize}
\end{theorem}

\subsection{Partial Derivatives and Jacobians}

\begin{exercise}
Let $z = e^{xy^2}$, $x = t \cos t$, $y = t \sin t$. Compute $\Dfrac{z}{t}$.
\end{exercise}
\begin{solution}
We use that $\Dfrac{z}{t} = \Dfrac{z}{x} \Dfrac{x}{t} + \Dfrac{z}{y} \Dfrac{y}{t}$.
\begin{align*}
\Dfrac{z}{t} &= y^2e^{xy^2}\cdot (\cos t -t\sin t) + 2xye^{xy^2} \cdot (\sin t + t \cos t) \\
&=t^2 \sin^2 t e^{t^3 \cos t \sin^2t}(\cos t -t\sin t) + 2xye^{t^3 \cos t \sin^2t} \cdot (\sin t + t \cos t)
\end{align*}
\end{solution}

\begin{exercise}
If $x=2r-s$ and $y=r+2s$, find $\frac{\partial^2 U}{\partial y \partial x}$
in terms of derivatives with respect to $r$ and $s$.
\end{exercise}
\begin{solution}
Use the multivariable chain rule, we should really solve for $r$ and $s$ in terms of $x$ and $y$.
We have $5r = 2x + y$ and $5s = 2y - x$. Therefore
\[
\pfrac{U}{x} = \pfrac{U}{r} \pfrac{r}{x} + \pfrac{U}{s} \pfrac{s}{x}
= \frac{2}{5} \pfrac{U}{r} - \frac{1}{5} \pfrac{U}{s}.
\]
We now differentiate by $y$.
\begin{align*}
\frac{\partial^2 U}{\partial y \partial x} &= \pfrac{}{y} \left(\frac{2}{5} \pfrac{U}{r} - \frac{1}{5} \pfrac{U}{s} \right) \\
&= \pfrac{}{r}\left(\frac{2}{5}\frac{\partial^2 U}{\partial r^2} - \frac{1}{5}\frac{\partial^2 U}{\partial rs}\right) \cdot \pfrac{r}{y}
+ \pfrac{}{s}\left(\frac{2}{5}\frac{\partial^2 U}{\partial sr} - \frac{1}{5}\frac{\partial^2 U}{\partial s^2}\right) \cdot \pfrac{s}{y} \\
&= \left(\frac{2}{5}\frac{\partial^2 U}{\partial r^2} - \frac{1}{5}\frac{\partial^2 U}{\partial rs}\right) \cdot \frac{1}{5}
+ \left(\frac{2}{5}\frac{\partial^2 U}{\partial sr} - \frac{1}{5}\frac{\partial^2 U}{\partial s^2}\right) \cdot \frac{2}{5} \\
&= \frac{1}{25} \left(2 \frac{\partial^2 U}{\partial r^2} + 3\frac{\partial^2 U}{\partial rs}
- 2\frac{\partial^2 U}{\partial s^2}\right)
\end{align*}
\end{solution}

\subsection{Line and Surface Integrals}

\begin{df}
Given a vector field $F$, if $\oint_{C} F \cdot d \textbf{r} = 0$ for any 
closed curve $C$, then we say that $\textbf{F}$ is a \textbf{conservative field}.
\end{df}

We recall some nice theorems that help computation.

\begin{theorem}[Green's Theorem in the Plane]
Suppose $R$ is a closed plane region bounded by a simple plane closed convex curve in the $x$-$y$ plane.
Let $L, M$ be continuous functions of $x, y$ having continuous derivatives throughout $R$. Then:
\[
\oint_C (L dx + M dy) = \int_R \left( \frac{\partial M}{\partial x} - \frac{\partial L}{\partial y} \right) dx dy,  
\]
where $C$ is the boundary of $R$ described in the counter-clockwise (positive) sense.
\end{theorem}

\begin{theorem}
Let $\mathbf{A}$ be a vector field. Then, the following conditions are equivalent:
\begin{itemize}
\item $\int_C \mathbf{A} \cdot d\mathbf{r}$ is independent
of the path.
\item There exists a scalar function $\theta$ such that $\mathbf{A} = \nabla \theta$.
In this case, the integral is equal to $\theta(P) - \theta(Q)$, where $P$ and $Q$ denote the finishing and starting points.
\item $\nabla \times \mathbf{A} = 0$ ($\curl \mathbf{A}= 0$). If $C$ is a closed path, then 
this $\nabla \times \mathbf{A} = 0$ is equivalent to $\mathbf{A}$ being convervative.
\end{itemize}
\end{theorem}

\begin{df}[Differential Element of Surface Area]
The differential element of surface area is
\[
dS =
\left|
\frac{\partial \mathbf{r}}{\partial v_1}
\times
\frac{\partial \mathbf{r}}{\partial v_2}
\right|
\,dv_1\,dv_2.
\]
\end{df}

\begin{theorem}[Divergence Theorem]
If $\tau$ is the volume enclosed by a closed surface $S$ with unit outward normal
$\mathbf{n}$ and $\mathbf{A}$ is a continuously differentiable vector field throughout $\tau$,
then 
\[
\int_S \mathbf{A} \cdot \mathbf{n} dS = \int_{\tau} \divergence \mathbf{A} d\tau.
\]
\end{theorem}

\begin{theorem}[Stokes' Theorem]
Suppose $S$ is an open surface with a simple closed curve $C$ forming its boundary, and let $\textbf{A}$ be a vector 
field with continuous partial derivatives. Then:
\[
\oint_{C} \mathbf{A} \cdot d \mathbf{r} = \oint_S \curl \mathbf{A} \cdot \hat{\mathbf{n}} dS
\]
where the direction of the unit normal to $S$ and the sense of $C$ are related by a right-hand rule. 
That is, $\hat{\mathbf{n}}$ is in the direction a right-handed screw moves when turned in the direction of $C$.
\end{theorem}

\begin{exercise}
Evaluate
\[
\int_{(0,1)}^{(1,2)} (x^2-y) dx+(y^2+x) dy
\]
along:
\begin{enumerate}
\item a straight line from $(0,1)$ to $(1,2)$.
\item straight lines from $(0,1)$ to $(1,1)$ and then from $(1,1)$ to $(1,2)$.
\item the parabola $x=t$, $y=t^2+1$.
\end{enumerate}
\end{exercise}
\begin{solution}
1. The line adjoining these is $y = x + 1$. Then, $dy = dx$, and we just plug in:
\[
\int_0^1 (x^2 - (x + 1))dx + ((x+1)^2 + x)dx 
= \int_0^1 2x^2 + 2x 
= \frac{5}{3}.
\]

2. We do a similar thing, but split the integral piecewise. The first part is
\[
\int_0^1 (x^2 - 1)dx + (1 + x)\cdot 0
= -\frac{2}{3}.
\]
The next part is
\[
\int_1^2 (1 - y) \cdot 0 + (y^2 + 1)dy 
= \frac{10}{3}.
\]
Thus, the integral is equal to $\frac{8}{3}$ by linearity. \\

3. For this, we note that $dx = dt$ and $dy = 2t dt$. The time is from $0$ to $1$.
\[
\int_0^1 (t^2 - (t^2 + 1))dt + ((t^2+ 1)^2 + t) 2tdt
= \int_0^1 2t^5 + 4t^2 + 2t^2 + 2t - 1 dt
= 2.
\]
\end{solution}

\begin{exercise}
Evaluate $\int_C yds$ along the curve $C$ given by $y = 2\sqrt{x}$
from $x = 3$ to $x = 24$.
\end{exercise}
\begin{solution}
Recall that $ds$ is the arclength element. It is probably easiest to use that
\[
ds = \sqrt{(dx)^2 + (dy)^2} = \sqrt{1 + \left(\frac{dx}{dy}\right)^2}dx.
\]
We have $\frac{dy}{dx} = \frac{1}{\sqrt{x}}$, so $ds = \sqrt{1 + \frac{1}{x}}dx$.
Thus, the integral is as follows:
\[
\int_3^{24} 2\sqrt{x} \cdot \sqrt{1 + \frac{1}{x}} dx
= 2\int_3^{24} \sqrt{x + 1} dx
= \frac{4}{3}(x + 1)^{\frac{3}{2}}\bigg|_3^{24}
= 156
\]
\end{solution}

\begin{exercise}
Verify Green's theorem in the plane for
\[
\oint_C (2xy-x^2) dx+(x+y^2) dy,
\]
where $C$ is the closed curve of the region bounded by $y=x^2$ and $y^2=x$.
\end{exercise}
\begin{solution}
Note that this is just from the origin to $(1,1)=:P$ and back. From the origin to $P$, we have:
\[
\int_0^1 (2x^3 - x^2)dx + (x + x^4)2xdx
= \int_0^1 2x^5 + 2x^3 + x^2 dx
= \frac{7}{6}.
\]
From $P$ to the origin, we have as follows:
\[
\int_1^0 (2y^3 - y^4) 2ydy + (y^2 + y^2)dy
= \int_1^0 -2y^5 + 4y^4 + 2y^2 dy
= -\frac{17}{15}
\]
Thus, the original integeral is $\frac{1}{30}$. \\

On the other hand, the statement of Green's theorem gives us that the 
integral is equal to the following:
\[
\int_R (1-2x)dxdy.
\]
Now we just need to find the appropriate bounds. We have as follows:
\[
\int_{y = 0}^1 \int_{x = y^2}^{\sqrt{y}} (1-2x)dxdy 
= \int_0^1 \sqrt{y} - y - y^2 + y^4 dy
= \frac{1}{30}.
\]
Thus, true!
\end{solution}


\begin{exercise}
Prove that
\[
\mathbf{F} = (2xz^3+6y)\mathbf{i} +(6x-2yz)\mathbf{j}
+(3x^2z^2-y^2)\mathbf{k}
\]
is a conservative force field. Further, evaluate
\[
\int_C \mathbf{F}\cdot d\mathbf{r},
\]
where $C$ is any path from $(1,-1,1)$ to $(2,1,-1)$.
\end{exercise}
\begin{solution}
To show it is conservative, we show that the curl of $\mathbf{F}$ is zero.
\[
\nabla \times \mathbf{F} = 0
\]
The easiest way to do this is to find some potential function, where
$\mathbf{F} = \nabla \theta$. Integrating variable by variable, we have that
$\phi(x,y,z) = x^2z^3 + 6xy - y^2z + c$, where $c$ is some constant, works.
We should really find the constant $c$, but they cancel anyway. The integral is
equal to 
\[
\phi(2,1,-1) - \phi(1,-1,1) = 15.
\]
\end{solution}

\begin{exercise}
Evaluate $\oint_C (2x-y+4) dx+(5y+3x-6) dy$ around the triangle in the $xy$-plane with 
vertices $(0,0)$, $(3,0)$, and $(3,2)$, traversed in the counterclockwise direction.
\end{exercise}
\begin{solution}
Use Green's theorem in the plane. The desired integral becomes the following.
\[
\int_{x = 0}^3 \int_{y = 0}^{\frac{2}{3}x} (3 - (-1)) dydx
= 12.
\]
\end{solution}

\begin{exercise}
Evaluate
\[
\int_{(1,1)}^{(4,2)} (x+y)\,dx+(y-x)\,dy
\]
along:
\begin{enumerate}
\item the parabola $y^2=x$
\item a straight line
\item straight lines from $(1,1)$ to $(1,2)$ and then to $(4,2)$
\item the curve $x=2t^2+t+1$, $y=t^2+1$
\end{enumerate}
\end{exercise}
\begin{solution}
1. We just plug in simply, using $dx = 2ydy$.
\[
\int_{y = 1}^2 (y^2 + y)2ydy + (y - y^2)dy
= \int_1^2 2y^3 + y^2 + y dy 
= \frac{34}{3}
\]
2. The line connecting the two is $y = \frac{1}{3}(x - 1) + 1 = \frac{1}{3}x + \frac{2}{3}$.
Therefore, $dy = \frac{1}{3} dx$, and the integral is as follows:
\[
\int_{x = 1}^4 (x + \frac{1}{3}x + \frac{2}{3}) dx + (\frac{1}{3}x + \frac{2}{3} - x) \frac{1}{3}dx
= \int_{1}^4 \frac{10}{9}x + \frac{8}{9} dx
= 11
\]
3. We split up the integral. From $(1,1)$ to $(1,2)$, $dx = 0$ and from $(1,2)$
to $(4,2)$, $dy = 0$.
\begin{align*}
\int_{y = 1}^2 (y - 1)dy &= \frac{1}{2} \\
\int_{x = 1}^4 (x + 2)dx &= \frac{27}{2}
\end{align*}
Thus, the total is $14$. \\

4. We parametrize in terms of $t$. Note that the bound is $t = 0$ to $t = 1$.
Also, $dx = (4t + 1)dt$, $dy = 2t dt$.
\[
\int_{t = 0}^1 (3t^2 + t + 2)(4t + 1)dt + (-t^2 - t)(2t)dt
= \int_{t = 0}^1 10t^3 + 5t^2 + 9t + 2
= \frac{32}{3}.
\]
\end{solution}
