\begin{exercise}

\begin{exercise}
Suppose $X$ is compact, $Y$ is Hausdorff, and $f:X\to Y$ is a continuous bijection. Which must be true?
\begin{enumerate}[label=(\Alph*)]
    \item $f^{-1}$ need not be continuous.
    \item $f$ is a homeomorphism.
    \item $Y$ must be compact but $f^{-1}$ can be discontinuous.
    \item $X$ must be connected.
    \item $Y$ must be metric.
\end{enumerate}
\end{exercise}
\begin{solution}
(B). This is a result; a continuous bijection from a compact to Hausdorff has a continuous inverse.
The proof isn't bad, so we'll do this here. Note that the image is compact by continuity,
and surjectivity implies $Y$ is compact. Now, $f$ is a closed map because if you take
some closed $Z \subseteq X$, then it is compact (closed subset of compact is compact), so its image 
is compact, so closed in $Y$. Now, a closed bijeciton is equivalent to an open bijection,
so the inverse is continuous.
\end{solution}

\begin{exercise}
We call a group $G$ ``pleasant'' if it is abelian and every non-identity
element $g$ in $G$ has prime order. Up to isomorphism, how many pleasant
groups are there of order at most $25$?
\begin{enumerate}[label=(\Alph*)]
    \item $0$
    \item $9$
    \item $15$
    \item $25$
    \item $31$
    \item Infinitely many
\end{enumerate}
\end{exercise}
\begin{solution}
(C). By the structure theorem for finitely generated abelian groups, 
$G$ is a finite product of finite cyclic groups. If two primes are present,
say $C_p$ and $C_q$, then there is an element of $C_{pq}$, which is not good.
Thus, $G$ is of the form $(C_p)^k$. There are nine primes total, but for 
$p = 2,3,5$, we can have powers. For $2$, we can have $k \leq 4$, For $3$ and $5$, 
$k \leq 2$. This gives $15$ possibilities.
\end{solution}

\begin{exercise}
Let $f$ and $g$ be twice-differentiable real-valued functions defined on
$\R$. If $f'(x)>g'(x)$ for all $x>0$, which of the following
inequalities must be true for all $x>0$?
\begin{enumerate}[label=(\Alph*)]
    \item $f(x)>g(x)$
    \item $f''(x)>g''(x)$
    \item $f(x)-f(0)>g(x)-g(0)$
    \item $f'(x)-f'(0)>g'(x)-g'(0)$
    \item $f''(x)-f''(0)>g''(x)-g''(0)$
\end{enumerate}
\end{exercise}
\begin{solution}
(C). Consider $h(x) = f(x) - g(x)$. Note that $h'(x) > 0$ and is twice differentiable.
$h$ is strictly increasing for all $x > 0$, so continuity (and thus the mean 
value theorem) implies that $h(x) > h(0)$, for all $x > 0$. 
\end{solution}

\begin{exercise}
Let $f(x,y)=x^2-2xy+y^3$ for all real $x$ and $y$. Which of the following is true?
\begin{enumerate}[label=(\Alph*)]
    \item $f$ has all of its relative extrema on the line $x=y$.
    \item $f$ has all of its relative extrema on the parabola $x=y^2$.
    \item $f$ has a relative minimum at $(0,0)$.
    \item $f$ has an absolute minimum at $\left(\frac{2}{3},\frac{2}{3}\right)$.
    \item $f$ has an absolute minimum at $(1,1)$.
\end{enumerate}
\end{exercise}
\begin{solution}
(A). $f_x = 2x - 2y$ and $f_y = -2x + 3y^2$. Thus, critical points are 
$P:=(0,0)$ and $Q:=\left( \frac{2}{3}, \frac{2}{3}\right)$. We now check the Hessian.
The Hessian has determinant $12y - 4$. At $P$, we have a saddle. At $Q$, the 
determinant is positive and $f_{xx}$ is positive, so we have a relative minima.
\end{solution}

\begin{exercise}
Say we have a triangle with sides $1,r,s$, with the angle between sides with 
lengths $1$ and $r$ being $110^{\circ}$. Here, as $r$ and $s$ increase, 
the length of the third side of the triangle remains $1$ and the measure 
of the obtuse angle remains $110^\circ$. What is
of the triangle remains $1$ and the measure of the obtuse angle remains
$110^\circ$. What is
\[
\lim_{\substack{s\to\infty\\r\to\infty}}(s-r)?
\]

\begin{enumerate}[label=(\Alph*)]
    \item $0$
    \item A positive number less than $1$
    \item $1$
    \item A finite number greater than $1$
    \item $\infty$
\end{enumerate}
\end{exercise}
\begin{solution}
(B). Use law of cosines to get
\[
s^2 = 1 + r^2 - 2r \cos (110^{\circ}).
\]
From this, we can deduce that 
\[
s - r = \frac{\frac{1}{r} - 2 \cos(110^{\circ})}{\frac{s}{r} + 1}.
\]
Now, we just need to determine the limit of $\frac{s}{r}$, but this is $1$, because
the law of cosines tell us that the limit of $\frac{s^2}{r^2}$ is $1$ (and the square 
root is continuous on positive reals). Precisely, the limits is $\cos(110^{\circ})$ then.
\end{solution}

\begin{exercise}
Let $\mathbf{F}$ be a constant unit force that is parallel to the vector
$(-1,0,1)$ in $xyz$-space. What is the work done by $\mathbf{F}$ on a
particle that moves along the path given by
$(t,t^2,t^3)$ between time $t=0$ and time $t=1$?

\begin{enumerate}[label=(\Alph*)]
    \item $\displaystyle -\frac{1}{4}$
    \item $\displaystyle -\frac{1}{4\sqrt{2}}$
    \item $0$
    \item $\sqrt{2}$
    \item $3\sqrt{2}$
\end{enumerate}
\end{exercise}
\begin{solution}
(C). Note that work is $\int \mathbf{F} \cdot d \mathbf{r}$.
We use a parametrization, so this is actually 
$\int \mathbf{F}(\mathbf{r}(t)) \cdot \mathbf{r}'(t) dt$.
We get
\[
\int_0^1 \frac{1}{\sqrt{2}} (-1,0,1) \cdot (1, 2t, 3t^2) dt 
= \int_0^1 \frac{3t^2 - 1}{\sqrt{2}} dt
= 0.
\]
Note that in the case of constant force (which is our case), we can 
just do $\mathbf{F} \cdot \nabla \mathbf{r}$.
\end{solution}

\begin{exercise}

functions on the interval $(0,\infty)$, then
\[

\]
The following argument is suggested as a proof of this theorem.

\begin{enumerate}
    \item By the Mean Value Theorem, there is a $c_1$ in the interval
    $(1,2)$ such that
    \[
    f'(c_1)
    =
    \frac{f(2)-f(1)}{2-1}
    =
    f(2)-f(1)>0.

    \[
    \frac{f(x)-f(2)}{x-2}=f'(c_x).
    \]

    \item For each $x>2$,
    \[
    \frac{f(x)-f(2)}{x-2}
    =
    f'(c_x)>f'(c_1)
    \]
    since $f'$ is strictly increasing.

    \item For each $x>2$,
    \[

(A). Note that the only cautious step may be the usage of the MVT, but

    \item 2 only

\end{solution}

\begin{exercise}
Let
\[
F(x)=\int_{\sin x}^{\cos x} e^{t^2+xt}\,dt.
\]
Then $F'(0)=$

\begin{enumerate}[label=(\Alph*)]
    \item $e\cos 1-2$
    \item $-2+e$
    \item $\dfrac{e}{2}-\dfrac{3}{2}$
    \item $\dfrac{e}{2}+\dfrac{1}{2}$
    \item $\dfrac{e}{2}-\dfrac{1}{2}$
\end{enumerate}
\end{exercise}
\begin{solution}
(C). We use Leibniz; $F'(x)$ is:
\[
F'(x) = e^{\cos^2x + \cos x t}\cdot (-\sin x) - e^{\sin^2x + \sin x t} \cdot ( \cos x) +

Setting $x = 0$, we have that
\[
F'(0) = 0 - 1 + \int_0^1 te^{t^2} dt 
= \frac{1}{2}(e - 1) - 1.

\end{solution}

\begin{exercise}
Which functions are entire:
\begin{enumerate}
    \item $f(z)=2^{z^2}$
    \item $g(z)=z^{2z}\sin z$
    \item $h(z)=\displaystyle\sum_{i=0}^{\infty}z^i$
\end{enumerate}
\begin{enumerate}[label=(\Alph*)]
    \item 1, 2 only
    \item 1 only
    \item 1 and 3 only
    \item 2 only
    \item 2 and 3 only
\end{enumerate}
\end{exercise}
\begin{solution}
(B). For 1, $2^{z^2} = e^{z^2 \cdot \ln 2}$. This is a composition of entire functions, so 
is entire. Next, $z^{2z} = e^{2z \ln z}$ is not entire because $\ln z$ is not entire (it requires

\end{solution}

\begin{exercise}
Which of the following holomorphic functions must be constant functions:

\begin{enumerate}
    \item An entire function with positive real part.
    
    \item An entire function whose image does not intersect a line $L \subset \mathbb{C}$.
    
    \item An entire function such that $f(0)=0$ and
    \[
    \frac{f(z)}{z}\to 0
    \qquad\text{as } |z|\to\infty.
    \]
    \textit{(Hint: $f(z)/z$ has a removable singularity at $0$, so can be extended to an entire function.)}
    
    \item A holomorphic function whose range misses two points.
\end{enumerate}

\begin{enumerate}[label=(\Alph*)]
    \item 1, 2, 3, 4
    \item 2, 3, 4 only
    \item 1, 2 only
    \item 3 only
    \item 1, 2, 3 only
\end{enumerate}
\end{exercise}
\begin{solution}
(E). 1 is classic; take $g(z) = e^{f(z)}$, which is bounded and entire. By Liouville, it's constant. 
2 is true because we can apply an affine transformation to make this into the statemenet of 1. 
$g(z) := \frac{f(z)}{z}$ has a removable singularity, so can be extended to an entire function.
Now $g$ is bounded outside a large disk, and is also bounded inside it by maximum modulus principle 
(or EVT more simply). Thus, Liouville says that $g$ is constant, but then this constant must be zero. Thus, $f = 0$.

\end{solution}

\begin{exercise}
Let $R$ be a ring. Which of the following are always true?
\begin{enumerate}
    \item The set of idempotents
    \[
    \{x\in R\mid x^2=x\}
    \]
    is a subring.

    \item If $x^2=x$ for all $x\in R$, then $R$ is commutative.

    \item The only idempotents of an integral domain $R$ are $0$ and $1$.
\end{enumerate}
\begin{enumerate}[label=\alph*)]
    \item 1 only
    \item 2 only
    \item 3 only
    \item 2 and 3 only
    \item None are true.
\end{enumerate}
\end{exercise}
\begin{solution}
(D). 1 is not true if we take $\Z$. 3 is clearly true. 2 is also true:
\[
2a = (a + a)^2 = 4a^2 = 4a \implies 2a = 0.
\]
Next,
\[
a + b = (a + b)^2 = a^2 + ab + ba + b^2 = a + ab + ba + b \implies ab = ba.
\]
In the second line, we really get $ab = -ba$, but $b = -b$, so we get commutativity.
\end{solution}

\begin{exercise}

\begin{enumerate}
    \item There is a function $f:\mathbb{R}^3\to\mathbb{R}$ such that
    \[
    \operatorname{grad} f
    =
    \langle y,x+z\cos y,\sin y\rangle.
    \]

    \item There is a vector field $\vec{F}:\mathbb{R}^3\to\mathbb{R}^3$ such that
    \[
    \operatorname{curl}F
    =
    \langle z,y,x\rangle.
    \]

    \item If $u,v:\mathbb{R}\to\mathbb{R}^3$ satisfy
    \[
    u'=v-u,
    \qquad
    v'=v+u,
    \]
    then $u\times v$ is constant.
\end{enumerate}

\begin{enumerate}[label=(\Alph*)]
    \item 1, 2, and 3
    \item 2 and 3 only
    \item 1 and 3 only
    \item 2 and 3 only

\[
x=t^2+2t,
\qquad
y=3t^4+4t^3,
\]
for all $t>0$. The value of $\dfrac{d^2y}{dx^2}$ at the point $(8,80)$ is

\begin{enumerate}[label=(\Alph*)]
    \item $4$
    \item $24$
    \item $32$
    \item $96$
    \item $192$
\end{enumerate}
\end{exercise}
\begin{solution}
(A). We note that 
\[
\Dfrac{y}{x} = \frac{\Dfrac{y}{t}}{\Dfrac{x}{t}} = 6t^2.
\]
Next, we note that 
\[
\frac{d^2y}{dx^2} = \Dfrac{}{t}\left(\Dfrac{y}{x}\right) \cdot \Dfrac{t}{x}
= \Dfrac{}{t}\left(\Dfrac{y}{x}\right) \cdot \frac{1}{\Dfrac{x}{t}}.
\]
Thus, $\frac{d^2y}{dx^2} = \frac{12t}{2t + 2}$. At $t = 2$, this yields $4$.
\end{solution}

\begin{exercise}
Let $f$ and $g$ be functions of a real variable such that
\[
g(x)=\int_0^x f(y)(y-x)\,dy
\]
for all $x$. If $g$ is three times continuously differentiable, what is the greatest integer $n$ for which $f$ must be $n$ times continuously differentiable?
\begin{enumerate}[label=(\Alph*)]
    \item $1$
    \item $2$
    \item $3$
    \item $4$
    \item $5$
\end{enumerate}
\end{exercise}
\begin{solution}
(A). By Lebniz rule, we have
\[
g'(x) = f(x)(x-x)\cdot 1 - f(0)(0-x)\cdot 0 + \int_0^x -f(y) dx.
\]
Thus, 
\[
g''(x) = -f(x).
\]
$g$ is three time differentiable, so we need $f'$ to exist at the minimum.
\end{solution}

\begin{exercise}
If $a$ and $b$ are positive numbers, what is the value of
\[
\int_0^\infty
\frac{e^{ax}-e^{bx}}
{(1+e^{ax})(1+e^{bx})}\,dx?
\]

\begin{enumerate}[label=(\Alph*)]
    \item $0$
    \item $1$
    \item $a-b$
    \item $(a-b)\log 2$
    \item $\displaystyle \frac{a-b}{ab}\log 2$
\end{enumerate}
\end{exercise}
\begin{solution}
(E). Note that the integrand is a difference of $\frac{1}{1 + e^{bx}}$ and same but 
$b$ replaced with $a$. Now,
\[
\int_0^{\infty} \frac{1}{1 + e^{bx}} dx 
= \int_0^{\infty} \frac{e^{-bx}}{1 + e^{-bx}} dx 
= -\frac{1}{b}\ln|1 + e^{-bx}| \Bigg|_0^{\infty}
= \frac{1}{b} \ln (2).
\]
Thus, it is (E).
\end{solution}

\begin{exercise}
For each positive integer $n$, let $x_n$ be a real number in the open interval
$\left(0,\frac{1}{n}\right)$. Which of the following statements must be true?

\begin{enumerate}
    \item $\lim_{n\to\infty} x_n=0$.
    \item If $f$ is a continuous real-valued function defined on $(0,1)$, then
    $\{f(x_n)\}_{n=1}^{\infty}$ is a Cauchy sequence.
    \item If $g$ is a uniformly continuous real-valued function defined on $(0,1)$, then 
    $\lim_{n\to\infty}g(x_n)$ exists.
\end{enumerate}
\begin{enumerate}[label=(\Alph*)]
    \item 1 only
    \item 1 and 2 only
    \item 1 and 3 only
    \item 2 and 3 only
    \item 1, 2, and 3
\end{enumerate}
\end{exercise}
\begin{solution}
(C). 1 is true by squeeze theorem, since $\frac{1}{n} \rightarrow 0$. 2 is false. 
Consider $f(x) = \frac{1}{x}$. Then, $f(x_n) > n$, so the given set can't be Cauchy.
3 is true. Uniform continuity and 1 tell us that $|g(x_n) - g(x_m)|$ tend to zero, thus Cauchy.
By completeness, Cauchy implies convergence.
\end{solution}

\begin{exercise}
A real-valued function $f$ defined on $\mathbb{R}$ has the following property: For 
every positive number $\varepsilon$, there exists a positive number $\delta$ such that
\[
|f(x)-f(1)|\geq\varepsilon
\qquad\text{whenever}\qquad
|x-1|\geq\delta.
\]

This property is equivalent to which of the following statements about $f$?

\begin{enumerate}[label=(\Alph*)]
    \item $f$ is continuous at $x=1$.
    \item $f$ is discontinuous at $x=1$.
    \item $f$ is unbounded.
    \item $\displaystyle \lim_{|x|\to\infty}|f(x)|=\infty$.
    \item $\displaystyle \int_0^\infty |f(x)|\,dx=\infty$.
\end{enumerate}
\end{exercise}
\begin{solution}
(D).  This is precisely the definition of (D). Vaguely, we get this because this says that 
if the input is really far from $1$, then the output is really from $f(1)$.
\end{solution}

\begin{exercise}
A tank initially contains a salt solution of $3$ grams of salt dissolved in $100$ liters of water.
A salt solution containing $0.02$ grams of salt per liter of water is sprayed into the tank at a rate 
of $4$ liters per minute. The sprayed solution is continually mixed with the salt solution in the tank, 
and the mixture flows out of the tank at a rate of $4$ liters per minute. If the mixing is instantaneous, 
how many grams of salt are in the tank after $100$ minutes have elapsed?
\begin{enumerate}[label=(\Alph*)]
    \item $2$
    \item $2-e^{-2}$
    \item $2+e^{-2}$
    \item $2-e^{-4}$
    \item $2+e^{-4}$
\end{enumerate}
\end{exercise}
\begin{solution}
(E). If $S(t)$ denotes the salt in the tank at time $t$, then we have
\[
S' = 4(0.02) - 4\left(\frac{S}{100}\right).
\]
This gives $S(t) = 2 + e^{-\frac{t}{25}}$, using the initial conditions.
\end{solution}

\begin{exercise}
What is the value of the flux of the vector field $\mathbf F$, defined on
$\mathbb{R}^3$ by
\[
\mathbf F(x,y,z)=x\mathbf i+y\mathbf j+z\mathbf k,
\]
through the surface
\[
z=\sqrt{1-x^2-y^2}
\]
oriented with an upward-pointing normal vector field?

\begin{enumerate}[label=(\Alph*)]
    \item $0$
    \item $\dfrac{2\pi}{3}$
    \item $\pi$
    \item $\dfrac{4\pi}{3}$
    \item $2\pi$
\end{enumerate}
\end{exercise}
\begin{solution}
(E). Use the divergence theorem. The integral is equal to 
\[
\int_V \nabla \cdot \mathbf{F} dV
= \int_V 3 dV.
\]
Now, the volume is half a sphere of radius $1$, so it is $\frac{2}{3} \pi$.
Thus, (E).
\end{solution}
